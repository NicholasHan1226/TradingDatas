SHELL=/bin/bash
PATH=/opt/marketgraph/venv/bin:/usr/local/bin:/usr/bin:/bin:/usr/sbin:/sbin

# SharedSignals cron template - install under the production crontab when ready.
# Paths intentionally target /opt/investment/SharedSignals production runtime.

# Tushare P0 must support the trading cadence directly. Keep it separate from
# daily/reference tiers so 5-minute data is not blocked by longer daily jobs.
*/5 9-15 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P0_trading_5min

# Daily/reference tiers run in their natural windows instead of every 2 hours.
35 16 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P1_eod_daily
30 18 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P2_financial_daily
15 6 * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P3_reference_daily
30 6 * * * /opt/investment/SharedSignals/cron/refresh_industry_map.sh
45 8 * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P4_macro_daily

# HK/US daily data is Tushare-backed. Run after the US and HK close windows.
10 6,17 * * 1-6 /opt/investment/SharedSignals/cron/collectors.sh --tier P5_hk_us_daily

# Event/other tier is lighter than P0 but should stay fresher than daily.
*/30 * * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P6_other_daily

# Sync SQLite read model into DuckDB every 5 minutes.
*/5 * * * * /opt/investment/SharedSignals/cron/duckdb_sync.sh

# Run patrol health checks and heal when score falls below threshold.
*/10 * * * * /opt/investment/SharedSignals/cron/patrol.sh

# Run watchdog SLA checks, auto-restart path, and halt escalation loop.
*/5 * * * * /opt/investment/SharedSignals/cron/watchdog.sh
