# SharedSignals cron template - verified against production on 2026-07-04.
#
# This file is an install template. The repository root crontab.txt is the
# production snapshot; the live server state remains the final source of truth:
#   sudo -u marketgraph crontab -l
# Production uses one combined marketgraph user crontab for SharedSignals,
# TradingAgent, and MarketGraph. Do not install this single-system file directly
# over live crontab; merge it with the other two system crontab references.

SHELL=/bin/bash
PATH=/opt/marketgraph/venv/bin:/usr/local/bin:/usr/bin:/bin:/usr/sbin:/sbin

# Tushare P0 must support the trading cadence directly. Keep it separate from
# daily/reference tiers so 5-minute data is not blocked by longer daily jobs.
*/5 9-15 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P0_trading_5min

# CN futures 5-minute bars use a dedicated Futures universe and must not run
# through the A-share per-stock P0 loop. The post-midnight night session runs
# Tuesday-Saturday because it belongs to the prior trading night's session.
*/5 9-15,21-23 * * 1-5 /opt/investment/SharedSignals/cron/cn_futures_5min.sh
*/5 0-2 * * 2-6 /opt/investment/SharedSignals/cron/cn_futures_5min.sh
25 16 * * 1-5 /opt/investment/SharedSignals/cron/cn_futures_daily.sh

# Daily/reference tiers run in their natural windows instead of every 2 hours.
35 16 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P1_eod_daily
30 18 * * 1-5 /opt/investment/SharedSignals/cron/collectors.sh --tier P2_financial_daily
15 6 * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P3_reference_daily
30 6 * * * /opt/investment/SharedSignals/cron/refresh_industry_map.sh
45 8 * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P4_macro_daily
10 6,17 * * 1-6 /opt/investment/SharedSignals/cron/collectors.sh --tier P5_hk_us_daily
*/30 * * * * /opt/investment/SharedSignals/cron/collectors.sh --tier P6_other_daily

# Crypto and PM market freshness.
2-59/5 * * * * /opt/investment/SharedSignals/cron/crypto_collect.sh
*/5 * * * * /opt/investment/SharedSignals/cron/pm_collect.sh

# RSS/RSSHub is retired from production collectors. Do not add the old
# /opt/investment/RSSCollector hot/warm or RSSHub @reboot jobs back.

# Read model sync, patrol, and SLA watchdog.
*/5 * * * * /opt/investment/SharedSignals/cron/duckdb_sync.sh
*/10 * * * * /opt/investment/SharedSignals/cron/patrol.sh
*/5 * * * * /opt/investment/SharedSignals/cron/watchdog.sh
3-59/10 * * * * /opt/investment/SharedSignals/cron/health_sla.sh
17 * * * * /opt/investment/SharedSignals/cron/capability_scan.sh
